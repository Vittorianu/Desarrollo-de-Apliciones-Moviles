    private val KEY_LOGGED_USER_ID = stringPreferencesKey("logged_user_id")
    private val KEY_OFFLINE_MODE = booleanPreferencesKey("offline_mode")
    private val KEY_LAST_SYNC_TIME = stringPreferencesKey("last_sync_time")

    suspend fun guardarSesionUsuario(userId: String) {
        context.dataStore.edit { prefs ->
            prefs[KEY_LOGGED_USER_ID] = userId
        }
    }

    suspend fun cerrarSesion() {
        context.dataStore.edit { prefs ->
            prefs.remove(KEY_LOGGED_USER_ID)
        }
    }

    fun obtenerUsuarioLogueadoId(): Flow<String?> {
        return context.dataStore.data.map { prefs ->
            prefs[KEY_LOGGED_USER_ID]
        }
    }

    suspend fun guardarModoOffline(activo: Boolean) {
        context.dataStore.edit { prefs ->
            prefs[KEY_OFFLINE_MODE] = activo
        }
    }

    fun obtenerModoOffline(): Flow<Boolean> {
        return context.dataStore.data.map { prefs ->
            prefs[KEY_OFFLINE_MODE] ?: false
        }
    }

    suspend fun guardarUltimaSincronizacion(timestamp: String) {
        context.dataStore.edit { prefs ->
            prefs[KEY_LAST_SYNC_TIME] = timestamp
        }
    }

    fun obtenerUltimaSincronizacion(): Flow<String?> {

data class ItemCotizacion(
    val id: String,
    val descripcion: String,
    val unidad: String,
    val cantidad: Double,
    val precioUnitario: Double,
    val porcentajeDescuento: Double = 0.0, // 0 - 100
    val porcentajeImpuesto: Double = 19.0,  // e.g. 19% IVA or 0%
    val esExento: Boolean = false
) {
    fun calcularNeto(decimales: Int): BigDecimal {
        val qty = BigDecimal.valueOf(cantidad)
        val price = BigDecimal.valueOf(precioUnitario)
        val discFactor = BigDecimal.ONE.subtract(BigDecimal.valueOf(porcentajeDescuento / 100.0))
        return qty.multiply(price).multiply(discFactor).setScale(decimales, RoundingMode.HALF_UP)
    }

    fun calcularImpuesto(decimales: Int): BigDecimal {
        if (esExento || porcentajeImpuesto <= 0) return BigDecimal.ZERO.setScale(decimales, RoundingMode.HALF_UP)
        val neto = calcularNeto(decimales)
        val taxFactor = BigDecimal.valueOf(porcentajeImpuesto / 100.0)
        return neto.multiply(taxFactor).setScale(decimales, RoundingMode.HALF_UP)
    }

    fun calcularTotalItem(decimales: Int): BigDecimal {
        return calcularNeto(decimales).add(calcularImpuesto(decimales)).setScale(decimales, RoundingMode.HALF_UP)
    }
}

data class Cotizacion(
    val id: String,
    val folio: String, // e.g. "COT-2026-001" or temp ID
    val empresaId: String,
    val usuarioPropietarioId: String,
    val clienteId: String,
    val clienteNombre: String,
    val contactoEmail: String,
    val fechaCreacion: LocalDate = LocalDate.now(),
    val fechaVencimiento: LocalDate = LocalDate.now().plusDays(30),
    val moneda: Moneda = Moneda.CLP,
    val items: List<ItemCotizacion> = emptyList(),
    val valorConversionCLP: Double = 1.0, // Valor indicador si USD u UF
    val fechaIndicador: LocalDate = LocalDate.now(),
    val fuenteIndicador: String = "Banco Central / Mindecon",
    val estadoComercial: EstadoCotizacion = EstadoCotizacion.BORRADOR,
    val estadoSync: EstadoSincronizacion = EstadoSincronizacion.SINCRONIZADO,
    val observaciones: String = ""
) {
    val subtotalNeto: BigDecimal
        get() = items.fold(BigDecimal.ZERO) { acc, item -> acc.add(item.calcularNeto(moneda.decimales)) }

    val totalImpuestos: BigDecimal
        get() = items.fold(BigDecimal.ZERO) { acc, item -> acc.add(item.calcularImpuesto(moneda.decimales)) }

    val totalCotizacion: BigDecimal
        get() = subtotalNeto.add(totalImpuestos)

    val equivalenciaCLP: BigDecimal
        get() {
            if (moneda == Moneda.CLP) return totalCotizacion.setScale(0, RoundingMode.HALF_UP)
            return totalCotizacion.multiply(BigDecimal.valueOf(valorConversionCLP)).setScale(0, RoundingMode.HALF_UP)
        }
}

@Composable
fun ValidatedTextField(
    value: String,
    onValueChange: (String) -> Unit,
    label: String,
    modifier: Modifier = Modifier,
    error: String? = null,
    keyboardOptions: KeyboardOptions = KeyboardOptions.Default,
    visualTransformation: VisualTransformation = VisualTransformation.None,
    readOnly: Boolean = false,
    singleLine: Boolean = true,
    trailingIcon: @Composable (() -> Unit)? = null
) {
    OutlinedTextField(
        value = value,
        onValueChange = onValueChange,
        label = { Text(text = label) },
        isError = error != null,
        readOnly = readOnly,
        singleLine = singleLine,
        keyboardOptions = keyboardOptions,
        visualTransformation = visualTransformation,
        trailingIcon = {
            if (error != null) {
                Icon(
                    imageVector = Icons.Default.Warning,
                    contentDescription = "Error",
                    tint = MaterialTheme.colorScheme.error
                )
            } else {
                trailingIcon?.invoke()
            }
        },
        supportingText = {
            error?.let {
                Text(text = it, color = MaterialTheme.colorScheme.error)
            }
        },
        modifier = modifier.fillMaxWidth()
    )
}

    content: @Composable () -> Unit
) {
    val colorScheme = when {
        dynamicColor && Build.VERSION.SDK_INT >= Build.VERSION_CODES.S -> {

        colorScheme = colorScheme,
        typography = Typography,
        content = content
    )

    }

            dataStore.guardarModoOffline(nuevoEstado)
        }

        }

        val state = _uiState.value
        val eErr = if (state.email.isBlank()) "El correo es obligatorio" else if (!state.email.contains("@")) "Correo invÃ¡lido" else null
        val cErr = if (state.clave.length < 4) "La contraseÃ±a debe tener al menos 4 caracteres" else null

        if (eErr != null || cErr != null) {
            _uiState.update { it.copy(emailError = eErr, claveError = cErr) }
            return false
        }

        _uiState.update { it.copy(esCargando = true) }

package com.example.android_16.viewmodel

import java.math.BigDecimal

data class ResumenIndicadores(
    val totalBorradores: Int = 0,
    val totalEnviadas: Int = 0,
    val totalAceptadas: Int = 0,
    val totalRechazadas: Int = 0,
    val totalFacturadas: Int = 0,
    val tasaAceptacionPorcentaje: Double = 0.0,
    val montoAceptadoFacturadoCLP: BigDecimal = BigDecimal.ZERO,
    val montoAceptadoFacturadoUSD: BigDecimal = BigDecimal.ZERO,
    val montoAceptadoFacturadoUF: BigDecimal = BigDecimal.ZERO
)

class IndicadoresViewModel : ViewModel() {

    private val repository = LocalDatabaseRepository

    val cotizaciones: StateFlow<List<Cotizacion>> = repository.cotizaciones

    fun calcularIndicadores(lista: List<Cotizacion>): ResumenIndicadores {
        val borradores = lista.count { it.estadoComercial == EstadoCotizacion.BORRADOR }
        val enviadas = lista.count { it.estadoComercial == EstadoCotizacion.ENVIADA }
        val aceptadas = lista.count { it.estadoComercial == EstadoCotizacion.ACEPTADA }
        val rechazadas = lista.count { it.estadoComercial == EstadoCotizacion.RECHAZADA }
        val facturadas = lista.count { it.estadoComercial == EstadoCotizacion.FACTURADA }

        // Tasa de AceptaciÃ³n = (Aceptadas + Facturadas) / (Aceptadas + Facturadas + Rechazadas) * 100
        val numerador = (aceptadas + facturadas).toDouble()
        val denominador = (aceptadas + facturadas + rechazadas).toDouble()
        val tasa = if (denominador > 0) (numerador / denominador) * 100.0 else 0.0

        val aceptadasYFacturadas = lista.filter { 
            it.estadoComercial == EstadoCotizacion.ACEPTADA || it.estadoComercial == EstadoCotizacion.FACTURADA 
        }

        var sumCLP = BigDecimal.ZERO
        var sumUSD = BigDecimal.ZERO
        var sumUF = BigDecimal.ZERO

        for (cot in aceptadasYFacturadas) {
            when (cot.moneda) {
                Moneda.CLP -> sumCLP = sumCLP.add(cot.totalCotizacion)
                Moneda.USD -> sumUSD = sumUSD.add(cot.totalCotizacion)
                Moneda.UF -> sumUF = sumUF.add(cot.totalCotizacion)
            }
        }

        return ResumenIndicadores(
            totalBorradores = borradores,
            totalEnviadas = enviadas,
            totalAceptadas = aceptadas,
            totalRechazadas = rechazadas,
            totalFacturadas = facturadas,
            tasaAceptacionPorcentaje = tasa,
            montoAceptadoFacturadoCLP = sumCLP,
            montoAceptadoFacturadoUSD = sumUSD,
            montoAceptadoFacturadoUF = sumUF
        )
    }
}

    @Test
    fun testCalculoNetoImpuestoYTotalItemCLP() {
        // Item: qty 2, price 100,000 CLP, discount 10%, IVA 19%
        val item = ItemCotizacion(
            id = "1",
            descripcion = "Servicio Test",
            unidad = "UN",
            cantidad = 2.0,
            precioUnitario = 100000.0,
            porcentajeDescuento = 10.0,
            porcentajeImpuesto = 19.0,
            esExento = false
        )

        // Net = 2 * 100000 * 0.9 = 180000 CLP
        val neto = item.calcularNeto(Moneda.CLP.decimales)
        assertEquals(BigDecimal("180000"), neto)

        // Tax = 180000 * 0.19 = 34200 CLP
        val impuesto = item.calcularImpuesto(Moneda.CLP.decimales)
        assertEquals(BigDecimal("34200"), impuesto)

        // Total Item = 180000 + 34200 = 214200 CLP
        val totalItem = item.calcularTotalItem(Moneda.CLP.decimales)
        assertEquals(BigDecimal("214200"), totalItem)
    }

    @Test
    fun testCalculoDecimalUSD() {
        // USD uses 2 decimals with HALF_UP rounding
        val item = ItemCotizacion(
            id = "2",
            descripcion = "Software SaaS",

            cantidad = 3.0,
            precioUnitario = 125.55,
            porcentajeDescuento = 5.0,
            porcentajeImpuesto = 0.0,
            esExento = true
        )

        // Net = 3 * 125.55 * 0.95 = 357.8175 -> rounded to 2 decimals HALF_UP = 357.82 USD
        val neto = item.calcularNeto(Moneda.USD.decimales)
        assertEquals(BigDecimal("357.82"), neto)
    }
